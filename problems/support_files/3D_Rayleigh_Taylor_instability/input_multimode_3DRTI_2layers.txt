Application = "Navier-Stokes"

// Variables that begin with "input" are commonly changed by user.

// Flow Properties
input_R_u  = 8.314462618153240 //Universal Gas Constant
input_Re_0 = 11000.0            //Reynolds Number
p_i        = 100000.0          //Interface pressure 
T_0        = 300.0             //Background temperature

// Case Properties
input_A                     = 0.5                                                 // Atwood number
species_mass_sum            = 0.064                                               // Sum of species molar masses
species_mass_diff           = input_A * species_mass_sum                          // Difference between species molar masses
species_mass_1              = 0.5 * (species_mass_sum + species_mass_diff)        // Mass of species 1
species_mass_2              = species_mass_sum - species_mass_1                   // Mass of species 2
input_Ma                    = 0.75                                                // Mach Number
R_i                         = (input_R_u * 2) / (species_mass_1 + species_mass_2) // Interface gas constant
d_i                         = p_i / (R_i * T_0)                                   // Interface density
c_0                         = sqrt( p_i / d_i )                                   // Isothermal speed of sound
mmin                        = 4                                                   // Minimum mode
mmax                        = 12                                                  // Maximum mode
dominant_waven              = (mmin + mmax) / 2                                   // Dominant Wave Number
input_lambda                = 701.53278340668                                     // We use the lambda of single mode cases, then extend the domain.
input_width                 = input_lambda * 2                                    // Domain size in y direction (Here, the single mode case extended twice in y-direction)
input_lambda_0              = input_width / dominant_waven                        // Characteristic Length Scale
g                           = ((input_Ma * c_0)^2.0) / input_lambda_0             // Gravity
input_delta	            = (input_lambda_0)*0.04
input_eta_0                 = (input_lambda_0)*0.04                               // Characteristic length of interface
input_special_source_box_lo = (-0.75 * input_width), -100000, -100000             // Left sponge boundary
input_special_source_box_hi = ( 0.75 * input_width), 100000, 100000               // Right sponge boundary
input_sponge_rate           = 0.25
input_sponge_nu_fact        = 1.0
input_cfl                   = 0.2e0                                                       // Max cfl factor used in problem
input_cfl_init              = 0.2e0                                                       // Initial cfl factor
input_end_time              = 10.0 * sqrt((input_lambda_0 / g)*(1.0 / input_A))            // Final simulation time
input_max_levels            = 3                                                           // Max number of AMR levels
input_domain_boxes          = [(0, 0, 0), (127, 63, 63)]                                                // Lower and upper indices of computational domain
input_x_lo                  = (-input_width), 0.0, 0.0                                           // Lower end of computational domain
input_x_up                  = ( input_width), (input_width) ,(input_width)     // Upper end of computational domain
input_viz_dump_interval     = input_end_time / 200.0    	                                          // Interval to output viz files
input_stat_dump_interval    = input_end_time / 200.0	                                                  // Interval to output to statistics file

// Fluid Properties
species_nu1          = sqrt((g * input_width^3.0) / input_Re_0^2.0) // Kinematic viscosity of species 1
species_nu2          = species_nu1                                      // Kinematic viscosity of species 2
input_species_gamma1 = 1.4                                              // Gamma of species 1
input_species_gamma2 = input_species_gamma1                             // Gamma of species 2
species_R1           = (input_R_u/species_mass_1)                       // Specific gas constant of species 1
species_R2           = (input_R_u/species_mass_2)                       // Specific gas constant of species 2
species_c_p1         = species_R1 / (1-(1/input_species_gamma1))        // Cp of species 1
species_c_p2         = species_R2 / (1-(1/input_species_gamma2))        // Cp of species 2
input_species_Pr_1   = 1.0                                              // Prandtl number of species 1
input_species_Pr_2   = input_species_Pr_1                                     // Prandtl number of species 2

    // Name of project
    project_name = "3D smooth multi-mode Rayleigh-Taylor instability"

    num_species = 2

    flow_model = "FOUR_EQN_CONSERVATIVE"

            species_gamma = input_species_gamma1, input_species_gamma2

            has_gravity = TRUE
            gravity = g, 0.0, 0.0 //NTBC for M

            special_source_box_hi = input_special_source_box_hi[0], input_special_source_box_hi[1], input_special_source_box_hi[2]  //1403.0655668133600,  1000
            sponge_rate    = input_sponge_rate //2.5
	    sponge_nu_fact = input_sponge_nu_fact //1.0 
	    species_mass   = species_mass_1, species_mass_2 // NTBC for A
            species_gamma = input_species_gamma1, input_species_gamma2
            width = input_width
            delta = input_delta
            eta_0 = input_eta_0
        }

        // Equation of shear viscosity to use
        equation_of_shear_viscosity = "CONSTANT"

        Equation_of_shear_viscosity_mixing_rules
        {
            use_constant_kinematic_viscosity_and_ideal_gas_assumptions = TRUE
            R_u = input_R_u
        //    species_mu = 15.3719746, 14.18951501 //NTBC for A, M, Re Dyn Visc
            species_nu = species_nu1 , species_nu2 //NTBC for A, M, Re 
            species_M  = species_mass_1 , species_mass_2   //NTBC for A W1&W2 kg/mol
        }

        // Equation of bulk viscosity to use
        equation_of_bulk_viscosity = "CONSTANT"
        Equation_of_bulk_viscosity_mixing_rules
        {
            species_mu_v = 0.0, 0.0
            species_M    = species_mass_1 , species_mass_2  //NTBC for A
        }

        // Equation of mass diffusivity to use
        equation_of_mass_diffusivity = "CONSTANT"

        Equation_of_mass_diffusivity_mixing_rules
        {
	        species_D    = species_nu1, species_nu2 //NTBC M, Re, A	    
	    }

        // Equation of thermal conductivity to use
        equation_of_thermal_conductivity = "PRANDTL"

        Equation_of_thermal_conductivity_mixing_rules
        {
            species_c_p = species_c_p1 , species_c_p2 //NTBC for A  
            species_Pr  = input_species_Pr_1, input_species_Pr_2
            species_M   = species_mass_1 , species_mass_2  //NTBC for A
        //    species_mu  = 15.3719746, 14.18951501 //NTBC for A, Re, M
            species_nu = species_nu1, species_nu2 //NTBC for A, Re, M

            equation_of_shear_viscosity = "CONSTANT"
            use_constant_kinematic_viscosity_and_ideal_gas_assumptions = TRUE
            R_u = input_R_u
        }

	// Names of statistical quantities to output
        statistical_quantities = 
            "MIXING_WIDTH_X",
          //  "MIXING_WIDTH_X_MOL_F",
          //  "MIXING_WIDTH_X_VOL_F",
            "MIXEDNESS_X",
          //  "MIXEDNESS_X_MOL_F",
          //  "MIXEDNESS_X_VOL_F",
            "ENSTROPHY_INT",
          //  "VORTICITY_Z_SQ_INT",
        //  "BARO_TQ_Z_SQ_INT",
	    "BOUSS_MEAN_INHOMO_X",
	    "TKE_INT_HOMO_YZ",
	    "TKE_X_INT_HOMO_YZ",
	    "TKE_Y_INT_HOMO_YZ",
	    "TKE_Z_INT_HOMO_YZ",
	    "a1_MEAN_INHOMO_X",
	    "b_MEAN_INHOMO_X",
	    "R11_MEAN_INHOMO_X",
	    "R22_MEAN_INHOMO_X",
            "SCAL_DISS_RAT_INT",
          //  "SCAL_DIS_R_INT_MOL_F",
          //  "SCAL_DIS_R_INT_VOL_F",
            "NUM_INTEF_THICK",
            "NUM_CELLS",
            "WEIGHTED_NUM_CELLS",
            "INTERFACE_MIN_X",
          //  "INTERFACE_MIN_X_MOL_F",
          //  "INTERFACE_MIN_X_VOL_F",
            "INTERFACE_MAX_X"
          //  "INTERFACE_MAX_X_MOL_F",
          //  "INTERFACE_MAX_X_VOL_F"
    }

    // Convective flux reconstructor to use
    convective_flux_reconstructor = "DRP4"
    stencil_width = 13
    Convective_flux_reconstructor{}

    use_conservative_form_diffusive_flux = FALSE

    // Non-conservative diffusive flux divergence operator to use
    nonconservative_diffusive_flux_divergence_operator = "TENTH_ORDER"

    Nonconservative_diffusive_flux_divergence_operator{
    }

    Initial_conditions
    {
        gravity      = g, 0.0, 0.0  //NTBC for M
        species_mass = species_mass_1 , species_mass_2  // NTBC for A
        x_lo         = input_x_lo[0], input_x_lo[1], input_x_lo[2]      // Lower end of computational domain
        x_up         = input_x_up[0], input_x_up[1], input_x_up[2]      // Upper end of computational domain
	species_gamma = input_species_gamma1, input_species_gamma2
	width = input_width
	delta = input_delta
        eta_0 = input_eta_0
	m_min = mmin
	m_max = mmax
    }

    Boundary_data
    {
        boundary_face_xlo
        {
            boundary_condition = "FLOW"
        }
        boundary_face_xhi
        {
            boundary_condition = "FLOW"
        }
        // Set the boundary conditions for edges
        boundary_face_yhi
        {
            boundary_condition = "FLOW"
        }

        boundary_face_ylo
        {
            boundary_condition = "FLOW"
        }
        boundary_face_zhi
        {
            boundary_condition = "FLOW"
        }

        boundary_face_zlo
        {

        }

	boundary_edge_xlo_ylo

        // Set the boundary conditions for nodes
        
        boundary_node_xlo_ylo

	Special_boundary_data
        {

    }
